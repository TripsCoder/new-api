def poll(task_id, attempts=120):

function validate(c,response){
  if(!object(c)||c.schema_version!==1||!['ready','partial'].includes(c.status)||!Array.isArray(c.models))throw Error('目录版本或状态未核实');
  if(!Number.isFinite(c.updated_at)||c.updated_at<=0||!Number.isFinite(c.checked_at)||c.checked_at<=0)throw Error('目录时间未核实');
  const clientNow=Date.now()/1000,headerTime=Date.parse(response.headers.get('Date')||'')/1000;
  const offset=(Number.isFinite(headerTime)?headerTime:c.checked_at)-clientNow,serverAt=clientNow+offset;
  if(serverAt-c.updated_at<0||serverAt-c.updated_at>90)throw Error('目录已失效或更新时间异常');
  const current=c.models.filter(m=>object(m)&&m.kind==='video'),seen=new Set();
  for(const m of current){
    if(typeof m.model!=='string'||!m.model.trim()||seen.has(m.model)||!['available','paused','reference_only'].includes(m.status)||typeof m.accepting!=='boolean'||typeof m.disabled!=='boolean'||!['executable','upstream_reference','unknown'].includes(m.parameter_status)||!Array.isArray(m.parameters)||!Array.isArray(m.groups)||!Array.isArray(m.examples)||!Array.isArray(m.source_groups))throw Error('视频模型结构未核实');
    if(!m.parameters.every(p=>object(p)&&typeof p.name==='string')||!m.groups.every(g=>object(g)&&typeof g.name==='string'&&typeof g.accepting==='boolean'&&Array.isArray(g.tiers)&&g.tiers.every(t=>object(t)&&typeof t.published==='boolean'))||!m.examples.every(e=>object(e)&&typeof e.kind==='string')||(m.capabilities!==null&&(!object(m.capabilities)||!Array.isArray(m.capabilities.media_fields)||!m.capabilities.media_fields.every(f=>typeof f==='string'))))throw Error('视频参数或价格结构未核实');
    seen.add(m.model);
  }
  return {models:current,catalogStatus:c.status,updatedAt:c.updated_at,serverAt,observedAt:performance.now()};
}
async function refresh(){
  if(busy||document.hidden)return;
  busy=true;const run=++requestId;
  try{
    const response=await fetch('/api/public/prices',{cache:'no-store',credentials:'omit',signal:AbortSignal.timeout(10000)});
    if(!response.ok)throw Error('同步请求失败');
    const data=await response.json(),next=validate(data.model_catalog,response);
    if(run!==requestId)return;
    snapshot=next;models=next.models;fresh=true;notice='';
    clearTimeout(expiryTimer);expiryTimer=setTimeout(expire,Math.max(1,(90-age())*1000+10));
  }catch(error){
    if(run!==requestId)return;
    fresh=false;notice=['目录版本或状态未核实','目录时间未核实','目录已失效或更新时间异常','视频模型结构未核实','视频参数或价格结构未核实'].includes(error.message)?error.message:'实时同步失败';
    clearTimeout(expiryTimer);
  }finally{if(run===requestId){busy=false;render();}}
}
