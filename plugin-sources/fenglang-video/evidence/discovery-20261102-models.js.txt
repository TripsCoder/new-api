/**
 * 各档位对应的实际像素（仅用于界面提示，提交流程默认只传 resolution）
 *
 * 实测记录 —— 这张表用的是「拿 16:9 的短边当边长」的算法，对非 16:9 比例并不准。
 * 平台真正做的是：**保持 16:9 档位的像素总量，再按比例重新分配**。
 * 实测成片量出来的值：
 *   1:1 + 720p → 960x960   （本表记 720x720）
 *   4:3 + 720p → 1112x834  （本表记 960x720）
 * 规律：16:9 / 9:16 用标准分辨率，其余比例 k = ceil(sqrt(面积/(宽比×高比)))、
 * 边长取偶。按这条规则重算可同时命中上面两个实测值。
 *
 * 所以将来若要改成只传 size（文档中优先级最高的字段），
 * **必须先用上面这组真值替换本表**，否则面积只有一半多点，等于压低画质。
 */

  '16:9': { '480p': '854x480', '720p': '1280x720', '1080p': '1920x1080', '4k': '3840x2160', '768p': '1366x768', '2k': '2560x1440' },
  '9:16': { '480p': '480x854', '720p': '720x1280', '1080p': '1080x1920', '4k': '2160x3840', '768p': '768x1366', '2k': '1440x2560' },
  '1:1': { '480p': '480x480', '720p': '720x720', '1080p': '1080x1080', '4k': '2160x2160', '768p': '768x768', '2k': '1440x1440' },
  '4:3': { '480p': '640x480', '720p': '960x720', '1080p': '1440x1080', '4k': '2880x2160', '768p': '1024x768', '2k': '1920x1440' },
  '3:4': { '480p': '480x640', '720p': '720x960', '1080p': '1080x1440', '4k': '2160x2880', '768p': '768x1024', '2k': '1440x1920' },

  // 9:21 = 21:9 的竖幕镜像，档位像素一致、宽高对调

export function expectedSize(aspectRatio, resolution) {
  return SIZE_TABLE[aspectRatio]?.[resolution] || '';
}

/**
 * 提交 size 字段要用的真实像素值。
 *
 * 为什么不直接用 SIZE_TABLE：那张表对非 16:9 比例是错的（见上面的实测记录）。
 * 平台实际做法是「保持 16:9 档位的总像素量，再按比例重新分配，边长取偶」，
 * 这里就按这条规则现算：
 *   k = ceil(sqrt(16:9 档位总像素 / (宽比 × 高比)))，宽 = 宽比×k、高 = 高比×k，各自取偶。
 *
 * 自检（与实测值一致）：
 *   1:1 + 720p → 960x960 ✓　4:3 + 720p → 1112x834 ✓
 *
 * 16:9 / 9:16 直接返回标准分辨率（这两个比例的档位就是标准像素）。
 */
export function exactSize(aspectRatio, resolution) {
  const std = SIZE_TABLE[aspectRatio]?.[resolution];
  if (std && (aspectRatio === '16:9' || aspectRatio === '9:16')) return std;

  const base = SIZE_TABLE['16:9']?.[resolution];
  if (!base) return '';
  const [a, b] = String(aspectRatio).split(':').map(Number);
  if (!a || !b) return '';

  const area = base.split('x').map(Number).reduce((x, y) => x * y, 1);
  const k = Math.ceil(Math.sqrt(area / (a * b)));
  const even = (n) => Math.max(2, Math.round(n / 2) * 2);
  return `${even(a * k)}x${even(b * k)}`;
}

/** 别名归并（大小写不敏感）：老标识 → 规范标识；不是别名就原样返回 */
export function canonicalVideoModel(modelId) {
  const name = String(modelId ?? '').trim();
  if (!name) return '';
  const key = Object.keys(VIDEO_MODEL_ALIASES).find((k) => k.toLowerCase() === name.toLowerCase());
  return key ? VIDEO_MODEL_ALIASES[key] : name;
}

/** 档位名大小写不敏感查表（与服务端 wallet.js 同一口径） */
function pickTier(map, tier) {
  if (!map) return '';
  const key = String(tier ?? '').trim();
  if (!key) return '';
  if (map[key]) return map[key];
  const hit = Object.keys(map).find((name) => name.toLowerCase() === key.toLowerCase());
  return hit ? map[hit] : '';
}

/**
 * 对外型号 + 档位 → 真实型号名；映射表外的型号原样返回（老任务与直传型号照常可用）。
 *
 * 查表前先归并别名：老标识 `seedance2.5满血版` 与 `seedance-2.5` 必须算到同一台型号上，
 * 否则老任务/老短剧会把价算成「未计价」。
 */
export function realVideoModel(modelId, tier) {
  const canonical = canonicalVideoModel(modelId);
  const id = Object.keys(VIDEO_MODEL_ROUTES).find((m) => m.toLowerCase() === canonical.toLowerCase());
  return pickTier(VIDEO_MODEL_ROUTES[id], tier) || String(modelId ?? '');
}

/** 真实型号名 → 它所属的对外型号（老任务记录里存的是真实型号，展示时要认得出来） */
function exposedIdOf(modelId) {
  const name = String(modelId ?? '').toLowerCase();

/*
 * 「档位 + 比例 → 像素尺寸」的换算**不在前端做**（2026-10-25 改口径）。
 *
 * 起因：GPT Image 系与 Seedream 系**只认 `size` 里的宽高**，`aspect_ratio` 会被上游
 * 静默丢掉（实测 doubao 只发 `size: "2K"` → 出 2048x2048 方图），
 * 所以「选了 16:9、出来 1:1」。
 *
 * 换算放在服务端 `server/imageSizes.js` 里做（`withPixelSize`），提交前统一改写入参，
 * 这样**站内提交与开放 API 走同一条路**，不会出现「页面修好了、API 还是方图」。
 * 前端照旧只发档位字符串 + aspect_ratio，计费口径（按档位）也就完全不变。
 */

/** 按「档位 + 画面比例」取像素尺寸；取不到就退到该档的 1:1，再不行给空串。 */
export function pixelSizeOf(tier, ratio, card) {
  const table = (card && card.pixelSizes) || PIXEL_SIZES;
  const row = table[tier] || {};
  return row[ratio] || row['1:1'] || Object.values(row)[0] || '';
}

/**
 * 55 折那档的对外模型名后缀（用户 2026-10-30 定：「75 折用原名，55 折用 `-s`」）。
 * 与服务端 server/officialPricing.js 的 `OFFICIAL_SUFFIX_55` 必须一致。
 */
const OFFICIAL_SUFFIX_55 = '-s';

/**
 * 把重复的折扣后缀折叠成一个：`…-s-s` → `…-s`（与服务端 collapseOfficialSuffix 同一口径）。
 * 历史草稿 / 历史任务里可能存着重复后缀，显示与提交前都要过这一步，
 * 否则任务卡标题会显示成 `doubao-…-s-s`（2026-10-30 用户报的问题）。
 */
export function collapseOfficialModelSuffix(modelId) {
  const id = String(modelId ?? '').trim();
  return id.replace(/(?:-s){2,}$/i, OFFICIAL_SUFFIX_55);
}

/** 剥掉折扣后缀，得到上游/价目表认的模型名 */
function officialBaseId(modelId) {
  let id = collapseOfficialModelSuffix(modelId);
  while (id.toLowerCase().endsWith(OFFICIAL_SUFFIX_55)) {
    id = id.slice(0, -OFFICIAL_SUFFIX_55.length);
  }
  return id;
}

/** 对外标识 → 该走的折扣分组（重复后缀先折叠，`-s-s` 仍算 55 折） */
function officialScopeOf(modelId) {
  return collapseOfficialModelSuffix(modelId).toLowerCase().endsWith(OFFICIAL_SUFFIX_55)
    ? 'sd-official-55'
    : 'sd-official-75';
}

let officialPricing = null;

export function setOfficialPricing(value) {
  if (value && typeof value === 'object' && value.prices) officialPricing = value;
  return officialPricing;
}

/**
 * 这个标识是不是官方协议模型（创作台按它切到 Token 计价、按它分「站内模型 / SD-官转」两个页签）；
 * `-s` 后缀的也算。
 *
 * 2026-11-02 收紧：只认**官方价目快照里的那四个型号**（剥离 `-s` 后比对），
 * 不再拿服务端下发的 `pricing.official.prices` 的键当判据。原因：
 * 特稳（`seedance-2.5-stable`）与 72 折官方卡共用同一个上游型号
 * `doubao-seedance-2-5-260628`，价目里一旦混进站内卡的名字，它就会被算进「SD-官转」页签；
 * 点页签时 `state.model` 换到该组第一张卡 —— 于是「切到 72 折」实际提交的还是特稳。
 */
export function isOfficialModel(modelId) {
  const id = officialBaseId(modelId);
  return Boolean(id && Object.prototype.hasOwnProperty.call(OFFICIAL_PRICE_SNAPSHOT.prices, id));
}

/**
 * 官方模型某一档的计价参数：`{ unitPrice, perSecondTokens, discount }`；认不出返回 null。
 *
 * 折扣跟着**这个模型标识**走（原名 = 75 折、`-s` = 55 折），与这个标识绑死的那把 Key 一致，
 * 所以界面预估与服务端预扣是同一个数；最终以 `usage.total_tokens` 结算（多退少补）。
 */
/**
 * 画面比例 + 档位 → 输出像素（短边按档位取：480 / 720 / 1080 / 2160）。
 * 与服务端 `officialFrameSize` 同一套算法，改动要两边一起改。
 */
function officialFrameSize(ratio, formula, tier) {
  const key = String(tier ?? '').trim().toLowerCase();
  const shortSide = Number(formula?.shortSides?.[key]) || Number(formula?.shortSide) || 720;
  const [w, h] = String(ratio || '16:9')
    .split(':')
    .map((n) => Number(n));
  const rw = Number.isFinite(w) && w > 0 ? w : 16;
  const rh = Number.isFinite(h) && h > 0 ? h : 9;
  const even = (n) => Math.max(2, Math.round(n / 2) * 2);
  return rw >= rh
    ? { width: even((shortSide * rw) / rh), height: shortSide }
    : { width: shortSide, height: even((shortSide * rh) / rw) };
}

export function officialPriceOf(modelId, tier, ratio, withVideoRef = false) {
  const src = officialPricing || OFFICIAL_PRICE_SNAPSHOT;
  // 价目表按**上游模型名**存（`-s` 只是折扣后缀，不参与查价）
  const row = src.prices?.[officialBaseId(modelId)];
  if (!row) return null;
  const key = String(tier ?? '').trim().toLowerCase();
  const hit = Object.keys(row).find((t) => t.toLowerCase() === key);
  if (!hit) return null;
  const pair = row[hit];
  /*
   * 单价是 `[不含视频参考, 含视频参考]` 两列（2026-10-31）：带参考视频的单用第二列，
   * 上游本来就给这一档更低的价。单位是**元 / 百万 token**，估算出 token 后乘它算钱。
   */
  const unitPrice =
    Number(Array.isArray(pair) ? (withVideoRef && pair.length > 1 ? pair[1] : pair[0]) : pair) || 0;
  if (!(unitPrice > 0)) return null;
  /*
   * 每秒 token 用**官方估算公式**现算：(宽 × 高 × 帧率) ÷ 1024。
   * 短边按这一档取、宽高按比例推 —— 换了档位（1080p / 4k）或换了比例，这个数都跟着变。
   */
  const formula = src.tokenFormula || OFFICIAL_PRICE_SNAPSHOT.tokenFormula;
  const { width, height } = officialFrameSize(ratio, formula, hit);
  const perSecondTokens = Math.round((width * height * (Number(formula?.fps) || 24)) / 1024);
  // 单帧 token = 宽 × 高 ÷ 1024：参考图一张算一帧、参考音频一秒算一帧（与服务端一致）
  const frameTokens = (width * height) / 1024;
  /*
   * 折扣跟着**这个模型标识**走（用户 2026-10-30 定：原名 = 75 折、`-s` = 55 折），
   * 不再"取两档里最便宜的那个" —— 那样会把 55 折的价格摆到 75 折的模型上。
   */
  const scope = officialScopeOf(modelId);
  const discount = Number(src.discount?.[scope]) || (scope === 'sd-official-55' ? 0.48 : 0.72);
  return { unitPrice, perSecondTokens, frameTokens, discount };
}

/**
 * 官方模型这一单的预估（2026-10-31 用户定「按实际消耗的 Token」）：
 *
 *   token = (输出秒数 + 参考视频秒数) × 每秒 token
 *         + 参考图张数 × 单帧 token
 *         + 参考音频秒数 × 单帧 token
 *   金额  = token ÷ 100万 × 官方单价 × 该标识自己的折扣，四舍五入到分
 *
 * 与服务端 `estimateOfficialTokens` / `officialCostOf` 同一套式子，两边同源：
 * 提交时按这个数预扣，成功后由服务端按上游 `usage.total_tokens` 多退少补。
 *
 * 返回 `{ tokens, cost, perSecond, seconds, discount }`；认不出返回 null。
 */
export function officialEstimate({
  model,
  tier,
  seconds,
  ratio,
  inputVideoSeconds = 0,
  refImages = 0,
  inputAudioSeconds = 0,
  withVideoRef = false,
} = {}) {
  const videoRef = Boolean(withVideoRef) || Number(inputVideoSeconds) > 0;
  const info = officialPriceOf(model, tier, ratio, videoRef);
  if (!info) return null;
  const outputSecs = Math.max(1, Math.min(30, Math.round(Number(seconds) || 1)));
  const inputSecs = Math.max(0, Math.min(60, Number(inputVideoSeconds) || 0));
  const audioSecs = Math.max(0, Math.min(60, Number(inputAudioSeconds) || 0));
  const imageCount = Math.max(0, Math.min(30, Math.round(Number(refImages) || 0)));
  const tokens = Math.round(
    (inputSecs + outputSecs) * info.perSecondTokens + (imageCount + audioSecs) * info.frameTokens,
  );
  const perSecond =
    Math.round(((info.perSecondTokens / 1_000_000) * info.unitPrice * info.discount) * 10000) / 10000;
  return {
    tokens,
    cost: Math.round(((tokens / 1_000_000) * info.unitPrice * info.discount) * 100) / 100,
    perSecond,
    seconds: outputSecs + inputSecs,
    discount: info.discount,
  };
}

export function setPricing(pricing) {
  if (pricing && typeof pricing === 'object') pricingOverride = pricing;
  return pricingOverride;
}

/* ==================================================================== *
 * 算力：对外统一的计价单位，1 人民币 = 10 算力。
 * 服务端（余额 / 流水 / 扣费）仍以元存储与结算，界面显示一律折算成算力。
 * ==================================================================== */
export const CREDIT_PER_CNY = 10;

/** 元 → 算力（界面显示用；服务端下发 creditPerCny 时以后者为准） */
export function toCredits(yuan) {
  const per = Number(pricingOverride?.creditPerCny) > 0 ? Number(pricingOverride.creditPerCny) : CREDIT_PER_CNY;
  return Math.round((Number(yuan) || 0) * per * 100) / 100;
}

/** 元 → 算力文本（去掉多余的 0，直接拼单位） */

 * 最多 9 张参考图、一口价按条计费（价在 .env 的 CHANNEL_VIDEO_PRICE，当前 1.99 元/条）。

    maxRefAudios: 0,
    features: {

      watermark: false,
      seed: false,
      outputFormat: false,
      refAudio: false,
      refVideo: false,
    },
    isChannel: true,
  };
}

/*
 * 管理员临时下架的模型（2026-10-23 用户要求）。
 *
 * 状态存在服务端 ca_model_flags，随 `GET /api/pricing` 的 `retiredModels` 一起下发，
 * 由 app.js 在启动与每次拉价目时调 setModelStates() 注入。这里只存一份标识集合：
 *   · 普通用户：卡片直接不摆（visibleVideoModels 过滤掉）；
 *   · 管理员：卡片照旧摆出来，带「已下架」标记与「重新上架」按钮 ——
 *     不摆就没地方点回来。
 * 已有任务、短剧草稿里存着的老标识仍能回显：getModel / priceOf* 走的是 MODELS，
 * 只是不再出现在可选列表里。
 */
let retiredModels = new Set();

export function setModelStates(list) {
  const next = new Set();
  for (const id of Array.isArray(list) ? list : []) {
    const key = String(id || '').trim().toLowerCase();
    if (key) next.add(key);
  }
  retiredModels = next;
  return next;
}

/** 这个模型是不是被临时下架了 */
export function isModelRetired(id) {
  const key = String(id ?? '').trim().toLowerCase();
  return key ? retiredModels.has(key) : false;
}

/**
 * 对外价倍率：服务端下发的为准，拉不到时用快照 —— **两边都是 1**。
 * 2026-10-16 起定价没有倍率了，这个函数留着只为兼容：万一某台服务端还没更新、
 * 仍下发 4，界面会跟着显示成它那一套，不至于两边对不上。新口径下它等于 1。
 */
function markup() {
  const value = Number(pricingOverride?.markup);
  return Number.isFinite(value) && value > 0 ? value : PRICE_SNAPSHOT.markup;
}

/**
 * 折扣：服务端按**当前账号**下发的专属折扣（2026-10-19 起）。
 * 1 = 没折扣；0.8 / 0.85 / 0.9 / 0.92 / 0.95 / 0.98 是管理员单独设的几档。
 * 广场靠它决定要不要显示「专属X折」标签与划掉的原价：≥1 就只显示一个价。
 */
function discount() {
  const value = Number(pricingOverride?.discount);
  if (!Number.isFinite(value) || value <= 0) return PRICE_SNAPSHOT.discount;
  return Math.min(1, value);
}

/** 给「模型广场」用的折扣值：≥1 表示没打折 */
export function pricingDiscount() {
  return discount();
}

/** 网关原价 → 元：下发值优先（正常就是 1，网关按人民币计价） */
function upstreamRate() {
  const value = Number(pricingOverride?.upstreamToCny);
  return Number.isFinite(value) && value > 0 ? value : PRICE_SNAPSHOT.upstreamToCny;
}

/**
 * 视频单价（**元/秒**）：表里的价 × 换算（恒 1）× 倍率（恒 1）—— 就是用户实付价。
 * 名字里的「base」是历史原因（原先它指不含折扣的「基础价」）；现在广场不再有划掉的
 * 原价，展示与算钱用的是同一个数（日常算钱用 priceOfVideo 也行）。
 * 按条计费的型号不走这里（它没有「每秒」的概念），见 basePriceOfVideoTask。
 */
export function basePriceOfVideo(modelId, tier) {
  /*
   * sd-官key（2026-10-30）：官方协议按 Token 计费，价目表里没有它们的行。
   * 这里折算成「元/秒」给**还没接 perToken 的旧入口**用（短剧分镜、无限画布、模型广场）：
   *   每秒单价 = 每秒 tokens ÷ 100万 × 官方单价 × 折扣
   * 与创作台的预估是同一个公式，所以「单价 × 秒数」和那边的总额对得上。
   */
  const official = officialPriceOf(modelId, tier);
  if (official) {
    return (official.perSecondTokens / 1_000_000) * official.unitPrice * official.discount;
  }
  const table = pricingOverride?.video || PRICE_SNAPSHOT.upstreamPerSecond;
  /*
   * 查价两层，与后端 wallet.js 的 videoCost / videoPrices.js 的 videoUnitPrice 同口径：
   *   ① 先按**前台模型名**查（`seedance-2.5`、`ov-seedance-2.5`）—— 服务端那张定价表
   *      ca_video_prices 的行就是「前台模型 + 档位」，改价改的是这一层；先归并老标识别名。
   *   ② 查不到再按**真实型号名**查 —— 2026-10-16 定价表精简后这层只为「老页面缓存里
   *      还带着后台型号名」这种情况留的兼容；表里已不再有那些行，新缓存一律走 ①。
   *
   * 档位名大小写不敏感（与图片价目表同一套 pickName；服务端查表也是这个口径）。
   * 大小写没对上就判「未计价」的话，界面会显示 0 元而服务端照常扣费。
   */
  const canonical = canonicalVideoModel(modelId);
  /*
   * 第三层：传进来的是**真实型号名**（老任务、直传）时，先折回它所属的前台模型再查 ——
   * 与服务端 wallet.js 的 exposedModelOfReal 同一口径，免得界面显示 0 元、服务端照常扣。
   */
  const exposed = exposedIdOf(canonical);
  const price =
    pickName(table?.[canonical], tier) ??
    pickName(table?.[exposed], tier) ??
    pickName(table?.[realVideoModel(exposed || canonical, tier)], tier);
  return price == null ? null : price * upstreamRate() * markup();
}

/**
 * 视频基础价（**元/条**）：按条计费的型号用这个 —— 出片即收这一笔，与时长无关。
 * 表里没有就是按秒计费的型号，返回 null（调用方去查 basePriceOfVideo）。
 */
export function basePriceOfVideoTask(modelId, tier) {
  const table = pricingOverride?.videoTask || PRICE_SNAPSHOT.upstreamPerTask;
  // 与 basePriceOfVideo 同样的查法（前台模型名 → 历史真实型号折回 → 真实型号映射）
  const canonical = canonicalVideoModel(modelId);
  const exposed = exposedIdOf(canonical);
  const price =
    pickName(table?.[canonical], tier) ??
    pickName(table?.[exposed], tier) ??
    pickName(table?.[realVideoModel(exposed || canonical, tier)], tier);
  return price == null ? null : price * upstreamRate() * markup();
}

/**
 * 视频最终价（元/秒）：单价 × 折扣（恒 1）—— 与实际扣费同一个数，不打任何折。
 * 价目表里没有这一档时返回 null（界面显示「未计价」）。
 */
export function priceOfVideo(modelId, tier) {
  const base = basePriceOfVideo(modelId, tier);
  if (base == null) return null;
  /*
   * 官方协议模型（2026-10-30）：服务端扣费走 `exact`（金额已含渠道折扣、不叠账号折扣），
   * 所以这里也不能乘 `discount()`，否则界面预估会比实扣低一截。
   */
  if (officialPriceOf(modelId, tier)) return base;
  return base * discount();
}

/** 视频最终价（元/条）：口径同上，只是基价按「条」算 */
export function priceOfVideoTask(modelId, tier) {
  const base = basePriceOfVideoTask(modelId, tier);
  return base == null ? null : base * discount();
}

/** 档位 / 画质名大小写不敏感查表（服务端也是这么查的，两边口径一致） */
function pickName(table, key) {
  if (!table) return undefined;
  const k = String(key ?? '');
  if (table[k] !== undefined) return table[k];
  const hit = Object.keys(table).find((name) => name.toLowerCase() === k.toLowerCase());
  return hit ? table[hit] : undefined;
}

/** 图片单价（元/张，含画质加价；表里的数就是用户实付价） */
export function basePriceOfImage(modelId, tier, quality) {
  const src = pricingOverride?.image || IMAGE_PRICE_SNAPSHOT;
  const rate = pricingOverride?.usdToCny || IMAGE_PRICE_SNAPSHOT.usdToCny;
  const base = pickName((src.base || IMAGE_PRICE_SNAPSHOT.base)[modelId], tier);
  if (base == null) return null;
  const add = pickName((src.qualityAdd || IMAGE_PRICE_SNAPSHOT.qualityAdd)[modelId], quality) || 0;
  return (base + add) * rate * markup();
}

/** 图片最终价（元/张，含画质加价；折扣恒 1）。没有价目时返回 null */
export function priceOfImage(modelId, tier, quality) {
  const base = basePriceOfImage(modelId, tier, quality);
  return base == null ? null : base * discount();
}

 * 接口保留，调用方照旧。

  return price;

const roundTo = (n, m) => Math.max(m, Math.round(n / m) * m);

/**
 * 按「档位 + 比例」算出实际像素尺寸。
 * 官方建议一律用像素写 size（写档位字符串在部分模型上会落到更低档计价），
 * 所以这里把档位换算成像素，并按各模型的约束（倍数、边长、总像素上下限、最短边）夹紧。
 */
export function computeImageSize(model, tier, ratio) {
  const base = model?.tierBases?.[tier];
  const [rw, rh] = String(ratio || '1:1').split(':').map(Number);
  if (!base || !rw || !rh) return '';

  const c = model.constraints || {};
  const m = c.multiple || 8;
  const r = rw / rh;

  let w = roundTo(r >= 1 ? base : base * r, m);
  let h = roundTo(r >= 1 ? base / r : base, m);

  const scale = (k) => {
    w = roundTo(w * k, m);
    h = roundTo(h * k, m);
  };

  // 先满足下限，再满足上限
  if (c.minPixels && w * h < c.minPixels) scale(Math.sqrt(c.minPixels / (w * h)) + 0.001);
  if (c.minShortSide && Math.min(w, h) < c.minShortSide) {
    scale(c.minShortSide / Math.min(w, h) + 0.001);
  }
  if (c.maxSide && Math.max(w, h) > c.maxSide) scale(c.maxSide / Math.max(w, h));
  if (c.maxPixels && w * h > c.maxPixels) scale(Math.sqrt(c.maxPixels / (w * h)));

  return `${w}x${h}`;
}

/** 图片任务的数量上限 */
export function maxImagesOf(model) {
  return model?.features?.maxN || 1;
}

/**
 * 按调用名取视频模型。
 *
 * 老任务记录、老短剧项目里存的是**真实型号名**（a-sd2.5-720p、c-sd2.5-480p-1…），
 * 认不出来就会掉到 MODELS[0]，那样「历史参数回填」会悄悄变成另一个模型。
 * 所以这里多一层反查：真实型号名归到它所属的对外模型。
 */
export function getModel(id) {
  /*
   * 渠道卡不在 MODELS 里（见 CHANNEL_MODEL_ID 的说明），单独认一次 ——
   * 否则它在视频页会被 getModel 兜到 MODELS[0]，界面上高亮的是渠道卡、
   * 校验与提交却按另一张卡走，是要出事的。
   */
  if (String(id ?? '') === CHANNEL_MODEL_ID) return channelCard() || MODELS[0];
  // 老标识 / 大小写先归并一次：老草稿与老短剧项目里存的是 `seedance2.5满血版`，
  // 归并后能直接命中现在这张卡片，而不是靠 MODELS[0] 兜底
  const want = canonicalVideoModel(collapseOfficialModelSuffix(id));
  const direct = MODELS.find((m) => m.id === want);
  if (direct) return direct;
  const exposed = exposedIdOf(want);
  return MODELS.find((m) => m.id === exposed) || MODELS[0];
}

/**
 * 界面上要显示的**型号名**（任务卡片、任务详情、灵感作品卡、短剧分镜参数都走这里）。
 *
 * 任务里存的是提交时那个型号标识（对外整模型的标识是 `seedance-2.5`，
 * 直接摆出来就会跟模型卡片上的显示名（`Seedance 2.5`）对不上，所以统一在这里翻一次：
 *   · 对外型号标识 / 图片型号标识 → 卡片上的显示名；
 *   · 真实上游型号（换线前的老任务、隐藏线路）→ 归到它所属的对外显示名；
 *   · 实在认不出来 → 原样显示，宁可不翻译，也不要乱贴一个名字上去
 *     （注意不能用 getModel 兜底，它认不出来会返回 MODELS[0]，会把老任务标成在售型号）。
 */
export function modelLabel(id) {
  // 官方模型的历史标识可能带重复后缀（`doubao-…-s-s`）：先折叠，能正常翻成卡片名
  const want = collapseOfficialModelSuffix(id);
  if (!want) return '';
  // 老标识先归并到规范名：老任务里的 `seedance2.5满血版` 也要显示成卡片上的 `Seedance 2.5`
  const canonical = canonicalVideoModel(want);
  /*
   * 历史标识 `seedance2.5-备用`（2026-10-25 撤市，现在只是 `seedance-2.5` 的末位腿）：
   * 老任务里还存着这串，界面上不能冒出「备用」二字 —— 统一按主卡名显示。
   */
  if (canonical === 'seedance2.5-备用') return 'Seedance 2.5';
  let cardId = canonical;
  let imageId = IMAGE_MODEL_ALIASES[String(canonical).toLowerCase()] || canonical;
  let card = MODELS.find((m) => m.id === cardId) || IMAGE_MODELS.find((m) => m.id === imageId);
  /*
   * 尾巴上挂了一个 `-s`、但那不是官方折扣标识（2026-11-02）：
   * 站内卡被上游回执的池名污染过，历史任务里躺着 `seedance-2.5-stable-s` 这种串 ——
   * 它既不是 48 折卡、也认不出是哪张卡，于是原样吐在任务标题上。
   * 剥掉这一层再认一次，能认出来就按那张卡显示（宁可显示特稳，也不会露这种内部串）。
   */
  if (!card && /-s$/i.test(cardId)) {
    const base = officialBaseId(cardId);
    const baseImage = IMAGE_MODEL_ALIASES[String(base).toLowerCase()] || base;
    const baseCard = MODELS.find((m) => m.id === base) || IMAGE_MODELS.find((m) => m.id === baseImage);
    if (baseCard) {
      cardId = base;
      imageId = baseImage;
      card = baseCard;
    }
  }
  /*
   * 官方 key 的模型要跟站内按秒的同名 Seedance 分开（2026-10-30 用户要求「加个后缀 官key55折 / 官key75折」）：
   * 任务卡、任务详情、调用记录里一律显示成「Seedance 2.0 Mini · 官key48折」，
   * 站内按秒的那几条不带后缀。折数由标识决定（带 `-s` = 48 折、原名 = 72 折），
   * 只在**显示层**加，模型标识与计价都不动。
   */
  if (card) {
    if (!isOfficialModel(card.id)) return card.name;
    const tier = officialScopeOf(card.id) === 'sd-official-55' ? '48折' : '72折';
    return `${card.name} · 官key${tier}`;
  }
  const exposed = exposedIdOf(canonical);
  return MODELS.find((m) => m.id === exposed)?.name || want;
}

/**
 * 该模型支持的画面比例档位。
 * 模型声明了 ratios 就只返回这几档；没声明的沿用全部档位。
 * 界面（主表单、短剧默认值、分镜弹窗）一律走这里，别直接用 ASPECT_RATIOS，
 * 否则会给用户摆出上游根本不支持的画幅。
 */
export function ratiosOf(modelOrId) {
  const model = typeof modelOrId === 'string' ? getModel(modelOrId) : modelOrId;
  const allow = Array.isArray(model?.ratios) && model.ratios.length ? model.ratios : null;
  if (!allow) return ASPECT_RATIOS;
  const list = ASPECT_RATIOS.filter((r) => allow.includes(r.value));
  return list.length ? list : ASPECT_RATIOS;
}

export function modeLabel(id) {
  return MODES.find((m) => m.id === id)?.label || id;
}

export function resolutionLabel(model, value) {
  if (!value) return '—';
  /* 4K / 2K 这类档位本身就是「数字 + K」，不能再按 P 拼（2026-10-17 云上线带了 2K 档，
   * 以前只判 4K，2K 会被拼成「2P」） */
  if (/^\d+\s*k$/i.test(String(value).trim())) return `${parseInt(value, 10)}K`;
  return `${parseInt(value, 10)}P`;
}

/**
 * 当前模型在该模式下需要哪些素材。
 *
 * 数字是**槽位数量**（决定面板摆几个位置），另有 `refVideo` 表示**参考视频能不能提交** ——
 * 两者是分开的：Seedance 2.5 挂着 10 个槽位却不许提交，2.0 挂 3 个且可以提交。
 * 与素材面板的对接见 public/js/materials.js 的 setPlan / validate。
 */
/**
 * 参考音频 / 参考视频的**单段时长上限**（秒），按前台模型给：
 *   · 2026-10-28 用户要求：「fast 和 mini 的参考音频视频不能超过 15s」；
 *   · 2026-10-29 用户要求：「2.5 中 参考音频和视频 2-30 秒，不要超过 30 秒」。
 *
 * 这三张卡的成片时长本身就封顶 15 / 30 秒，上游对参考素材也按同一个数收；
 * 其它型号（2.0 / H3 / wan3）上游口径不同，素材面板照旧不限时长。
 *
 * 素材面板（materials.js）拿这个数当**单段上限**，同时也当**同类合计上限**
 * （视频合计 ≤ 这个数、音频合计 ≤ 这个数），单段下限统一 2 秒；
 * 最终提交前还有一道 mediaLimits.js 的复核（画布节点、站内提交共用）。
 * 时长由 materials.js 在挂上时用 `<video>/<audio>` 元数据读出来，
 * 读不到（跨域、格式坏）就不拦 —— 宁可让上游去判，也别误伤正常素材。
 */
const REF_MEDIA_MAX_SECONDS = {
  'seedance-2.5': 30,
  'seedance-2.0-fast': 15,
  'seedance-2.0-mini': 15,
  /*
   * sd-官key（2026-10-30）：官方协议的参考素材口径跟成片时长同源 ——
   * 2.5 上限 30 秒、2.0 系 15 秒；官方模型标识就是官方模型名，所以按 id 各挂一份。
   */
  'doubao-seedance-2-5-260628': 30,
  'doubao-seedance-2-5-260628-s': 30,
  'doubao-seedance-2-0-260128': 15,
  'doubao-seedance-2-0-260128-s': 15,
  'doubao-seedance-2-0-fast-260128': 15,
  'doubao-seedance-2-0-fast-260128-s': 15,
  'doubao-seedance-2-0-mini-260615': 15,
  'doubao-seedance-2-0-mini-260615-s': 15,
};

export function materialPlan(model, mode) {
  switch (mode) {
    case 'i2v':
      return { first: true, last: false, refImages: 0, refVideos: 0, refAudios: 0 };
    case 'firstlast':
      return { first: true, last: true, refImages: 0, refVideos: 0, refAudios: 0 };
    case 'reference':
      return {
        first: false,
        last: false,
        refImages: model.maxRefImages,
        refVideos: model.maxRefVideos,
        refAudios: model.features.refAudio ? model.maxRefAudios : 0,
        refVideo: model.features.refVideo !== false,
        // 参考素材单段时长上限（0 = 不限），见上面 REF_MEDIA_MAX_SECONDS
        refMediaMaxSeconds: REF_MEDIA_MAX_SECONDS[model.id] || 0,
      };
    default:
      return { first: false, last: false, refImages: 0, refVideos: 0, refAudios: 0 };
  }
}
